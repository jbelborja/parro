{{ $destination := .Destination }} {{ $isPDF := strings.HasSuffix (lower $destination) ".pdf" }} {{ $isRemote := or (strings.HasPrefix (lower $destination) "http://") (strings.HasPrefix (lower $destination) "https://") (strings.HasPrefix $destination "//") (strings.HasPrefix (lower $destination) "mailto:") (strings.HasPrefix (lower $destination) "tel:") }} {{ if and $isPDF (not $isRemote) }} {{ $unescaped := replace (replace $destination "%20" " ") "+" " " }} {{ $cleanDest := strings.TrimPrefix "./" $unescaped }} {{/* 1. Intentamos obtener el PDF con la ruta exacta tal cual se escribió */}} {{ $resource := .Page.Resources.GetMatch $destination }} {{/* 2. Intentamos con los espacios decodificados */}} {{ if not $resource }} {{ $resource = .Page.Resources.GetMatch $cleanDest }} {{ end }} {{/* 3. Intentamos solo con el nombre base del archivo */}} {{ if not $resource }} {{ $resource = .Page.Resources.GetMatch (path.Base $cleanDest) }} {{ end }} {{/* 4. Intentamos coincidencia con comodines alrededor del nombre base */}} {{ if not $resource }} {{ $resource = .Page.Resources.GetMatch (printf "*%s*" (path.Base $cleanDest)) }} {{ end }} {{/* 5. Si la página tiene un PDF en su bundle, lo usamos para que siempre lo encuentre */}} {{ if not $resource }} {{ $resource = .Page.Resources.GetMatch "*.pdf" }} {{ end }} {{/* 6.
